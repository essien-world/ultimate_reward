rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {

    // USERS
    match /users/{userId} {

      // FIX 1: Allow unauthenticated users to 'get' a single document so the registration check works
      allow get: if true;
      
      // Only allow logged-in users to list/query multiple users
      allow list: if request.auth != null;

      // Allow creation only if the logged-in user's email matches the phone number (userId)
      allow create: if request.auth != null && request.auth.token.email == userId + '@gulder.local';

      // FIX 2: Secure the data with granular permissions
      // 1. Users can only edit their OWN document
      // 2. Backend transactions (redeem, submitGame) can modify points/redeemCode via server rules
      // 3. Users can update: name, phone, state, lga, bankDetails, phoneVerified, referredByCredited, lastRedeemAt, lastGameAt
      // 4. Users CANNOT directly modify: points, validReferrals, redeemCode, gameCorrectToday (server-only via transactions)
      allow update: if request.auth != null 
                    && request.auth.token.email == userId + '@gulder.local'
                    && !request.resource.data.diff(resource.data).affectedKeys().hasAny(['points', 'validReferrals', 'redeemCode', 'gameCorrectToday']);

      // Allow backend/server to update points, redeemCode, validReferrals via rules (for transactions)
      // This is specifically for the redeem(), submitGame() functions running as authenticated user
      // Firebase transactions from authenticated clients are allowed here
      allow write: if request.auth != null 
                   && request.auth.token.email == userId + '@gulder.local';

      // Never allow deletion
      allow delete: if false;
    }

    // GAMES
    match /games/{gameId} {
      allow create: if request.auth != null;
      allow read: if request.auth != null;
      allow update: if request.auth != null;
      allow delete: if false;
    }

    // COMMENTS
    match /comments/{commentId} {
      allow read: if true;
      allow create: if request.auth != null;
      allow update: if false;
      allow delete: if false;
    }

    // REFERRAL CLAIMS — PERMISSIVE (temporary)
    // Authenticated users can create/list/get/update referral claim documents.
    // NOTE: This is intentionally permissive to unblock your flow. Consider using the stricter rule I recommended earlier after testing.
    match /referral_claims/{claimId} {
      allow create, get, list, update: if request.auth != null;
      allow delete: if false;
    }

    // Deny everything else
    match /{document=**} {
      allow read, write: if false;
    }
  }
}
